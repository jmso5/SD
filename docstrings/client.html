import grpc import index_pb2_grpc import index_pb2 from google.protobuf import empty_pb2 import json import sys import time

MAX_MESSAGE_LENGTH = 262144000 # tamanho máximo de mensagem

class Client: ““” - Classe Client que representa a comunicação de um cliente com a Gateway que vai realizar os “requests” do cliente através do gRCP. Estes requests vão ser posteriormente tratados pela Gateway e enviados para o resto do sistema. Recebe o IP e a porta do Gateway para estabelecer a comunicação. ““”

def __init__(self, sever_ip, server_port):
    """
    - Inicia o cliente criando um canal gRPC e um stub para se conectar com o Gateway que é utilizada como servidor.

    - É definido um tamanho máximo para as mensagens enviadas e recebidas para permitir a transmissão de grandes quantidades de dados. 
    Evitando erros relacionados com o tamanho das mensagens.
    """

    # permite enviar e receber mensagens grandes
    options = [
        ('grpc.max_send_message_length', MAX_MESSAGE_LENGTH),
        ('grpc.max_receive_message_length', MAX_MESSAGE_LENGTH),
    ]

    self.channel = grpc.insecure_channel(f"{sever_ip}:{server_port}", options=options) # cria canal gRPC com o gateway
    self.stub = index_pb2_grpc.GatewayStub(self.channel) # cria stub (cliente) -> putNew / takeNext

def submit_url(self, url):
    """
    - Método que submete um URL para o sistema através do Gateway.
    - Envia um request putNew para o Gateway com o URL a ser indexado.
    """

    try:
        request = index_pb2.PutNewRequest(url=url)
        self.stub.putNew(request) # chama o método RPC putNew do gateway
        print(f"URL {url} adicionado para indexação.")
        
    except grpc.RpcError as e:
        print(f"Não foi possível realizar o seu pedido. Tente novamente em 5 segundos.")
        time.sleep(5)

def statistics(self):
    """
    - Obtém estatísticas do sistema a partir do Gateway e retorna um
      dicionário simples (JSON-serializável) com as chaves:
      { "top_searches": [...], "barrel_stats": [{ "barrel_id": ..., "index_size": ..., "avg_response_time": ...}, ...] }
    """
    try:
        response = self.stub.getStatistics(empty_pb2.Empty())  # chama o método RPC getStatistics do gateway

        # Extrair top_searches (protobuf repeated -> list)
        top_searches = list(response.top_searches) if hasattr(response, "top_searches") else []

        # Extrair estatísticas dos barrels e converter para tipos primitivos
        barrel_stats = []
        for stat in getattr(response, "barrel_stats", []):
            barrel_id = stat.barrel_id if isinstance(stat.barrel_id, str) else str(stat.barrel_id)
            # rsplit para evitar partir IPv6 (opcional se quiseres só a porta)
            # display_id = barrel_id.rsplit(":", 1)[-1] if isinstance(barrel_id, str) and ":" in barrel_id else barrel_id

            barrel_stats.append({
                "barrel_id": barrel_id,
                "index_size": int(getattr(stat, "index_size", 0)),
                "avg_response_time": float(getattr(stat, "avg_response_time", 0.0))
            })

        return {
            "top_searches": top_searches,
            "barrel_stats": barrel_stats
        }

    except grpc.RpcError as e:
        #print(f"[Client] Erro ao obter estatísticas do Gateway: {e}")
        # Retornar estrutura vazia previsível para o websocket não falhar
        return {"top_searches": [], "barrel_stats": []}
    
    
def search_backlinks(self, url):
    """
    - Método que obtém os backlinks de um URL.
    - Apresenta a lista de URLs que apontam para o URL pedido.
    """

    try:
        response = self.stub.getBacklinks(index_pb2.PutNewRequest(url=url)) # chama o método RPC getBacklinks do gateway
        backlinks = response.backlinks # lista de urls que apontam para o url pedido

        if not backlinks:
            print(f"\nNenhum backlink encontrado para '{url}'.")
            return

        print(f"\nBacklinks para '{url}':")
        for backlink in backlinks:
            print(f"- {backlink}")
        return backlinks 

    except grpc.RpcError as e:
        print(f"Não foi possível realizar o seu pedido. Tente novamente em 5 segundos.")
        time.sleep(5)
